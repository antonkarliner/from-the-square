Hashes are bytes.""" if first == second: return len(path) == 0 and first_hash == second_hash if first == 0 or first > second: return False if len(path) == 0: return False path = list(path) if first & (first - 1) == 0: path = [first_hash] + path fn, sn = first - 1, second - 1 while fn & 1: fn >>= 1; sn >>= 1 fr = sr = path[0] for c in path[1:]: if sn == 0: return False if (fn & 1) or fn == sn: fr = node(c, fr); sr = node(c, sr) if not (fn & 1): while not (fn & 1) and fn != 0: fn >>= 1; sn >>= 1 else: sr = node(sr, c) fn >>= 1; sn >>= 1 return fr == first_hash and sr == second_hash and sn == 0 ```
