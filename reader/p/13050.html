Submitted for listing 65. Full source, no dependencies beyond the standard library. """Marrow prototype - device-agnostic behavioural recognition from ordinary input events. Fresh copy: save this message as marrow.py, then run: python3 marrow.py Reads/writes only local synthetic JSONL recordings. No capture, no network, no real data. Recording format (JSONL, one event per line), timestamps are integer milliseconds from session start, pointer positions are fractions of screen width/height in [0,1]: {"t":120,"kind":"move","x":0.412,"y":0.553} {"t":980,"kind":"click","x":0.412,"y":0.553} {"t":1010,"kind":"text","source":"typed|osk|dictated","ch":"a"} (ch "\b" = correction) style-a: mouse + physical keyboard. style-b: slow pointer + dictation/osk. Each style has an owner (rested and tired), a stranger, a mimic of the owner's habits, and a second stranger for the swap test. Threshold 3.0 is fixed before evaluation. """ import json, math, random from pathlib import Path THRESHOLD = 3.0 EPS = 1e-9 BASE = Path(__file__).parent / "marrow_data" STYLE = { "a": dict(jit=0.16, gap=95, dictated=0.0), "b": dict(jit=0.22, gap=14, dictated=0.85), } PROFILES = { "a-owner": ("a", 1.00, 1.0, 0.050), "a-tired": ("a", 0.78, 1.5, 0.047), "a-stranger": ("a", 1.18, 0.7, 0.003), "a-mimic": ("a", 0.85, 1.5, 0.100), "a-other": ("a", 0.66, 1.9, 0.001), "b-owner": ("b", 1.00, 1.0, 0.200), "b-tired": ("b", 0.75, 1.6, 0.180), "b-stranger": ("b", 1.25, 0.6, 0.020), "b-mimic": ("b", 0.80, 1.7, 0.450), "b-other": ("b", 0.60, 2.0, 0.010), } def generate(name, seed): style, sm, tm, corr = PROFILES[name] s, rng = STYLE[style], random.Random(seed) ev, t, x, y = [], 0, 0.5, 0.5 for i in range(90): t += max(8, int(rng.gauss(70 * sm * (1.6 if "tired" in name else 1), 18))) x = min(0.99, max(0.01, x + rng.gauss(0, 0.022 * s["jit"] * 4))) y = min(0.99, max(0.01, y + rng.gauss(0, 0.022 * s["jit"] * 4))) ev.append({"t": t, "kind": "move", "x": round(x, 4), "y": round(y, 4)}) if i % 15 == 0: ev.append({"t": t + 4, "kind": "click", "x": round(x, 4), "y": round(y, 4)}) span = t for i in range(400): src = "dictated" if rng.random() < s["dictated"] else ("osk" if rng.random() < 0.15 else "typed") ch = "\b" if rng.random() < corr else rng.choice("abcdefghij ") ev.append({"t": min(span, int(span * (i + 1) / 401 + rng.gauss(0, s["gap"] * tm / 6))), "kind": "text", "source": src, "ch": ch}) ev.sort(key=lambda e: e["t"]) return ev def features(ev): moves = [e for e in ev if e["kind"] == "move"] clicks = [e for e in ev if e["kind"] == "click"] texts = [e for e in ev if e["kind"] == "text"] speeds, turns, total = [], 0, 0.0 for a, b in zip(moves, moves[1:]): d = math.hypot(b["x"] - a["x"], b["y"] - a["y"]) dt = max(1, b["t"] - a["t"]) / 1000 speeds.append(d / dt) total += d for a, b in zip(speeds, speeds[1:]): if a > EPS and b > EPS and abs(math.log(b / a)) > 0.8: turns += 1 net = math.hypot(moves[-1]["x"] - moves[0]["x"], moves[-1]["y"] - moves[0]["y"]) if moves else 0 cgaps = [b["t"] - a["t"] for a, b in zip(clicks, clicks[1:])] tgaps = [b["t"] - a["t"] for a, b in zip(texts, texts[1:])] ss, tg, cs = sorted(speeds), sorted(tgaps), sorted(cgaps) inter = sum(any(abs(m["t"] - q["t"]) < 180 for q in texts[:6]) for m in moves) / max(1, len(moves)) return [ss[len(ss) // 2] if ss else 0, ss[int(len(ss) * 0.9)] if ss else 0, net / max(total, EPS), turns / max(1, len(moves) - 2), cs[len(cs) // 2] if cs else 0, sum(e["source"] == "dictated" for e in texts) / max(1, len(texts)), sum(e["ch"] == "\b" for e in texts) / max(1, len(texts)), tg[len(tg) // 2] if tg else 0, inter] def fit(rows): n = len(rows) mu = [sum(r[j] for r in rows) / n for j in range(len(rows[0]))] sd = [max(math.sqrt(sum((r[j] - mu[j]) ** 2 for r in rows) / n), 0.08 * abs(mu[j]), 0.002) for j in range(len(mu))] return mu, sd def score(vec, model): return max(abs(v - m) / s for v, m, s in zip(vec, *model)) def load(name, idx): return [json.loads(l) for l in (BASE / f"{name}-{idx}.jsonl").read_text().splitlines() if l] def main(): BASE.mkdir(exist_ok=True) for name in PROFILES: for i in range(8): p = BASE / f"{name}-{i}.jsonl" if not p.exists(): p.write_text("".join(json.dumps(e) + "\n" for e in generate(name, 1000 + i * 17 + sum(map(ord, name)) % 97))) print("recording format: JSONL ms / normalized x,y / text source labels; synthetic only") for style in ("a", "b"): own, tired, other = f"{style}-owner", f"{style}-tired", f"{style}-other" model = fit([features(load(who, i)) for who in (own, tired) for i in range(8)]) print(f"\n{style}: profile {own}+tired, threshold fixed at {THRESHOLD}") for who in PROFILES: if who.startswith(style + "-"): sc = [score(features(load(who, i)), model) for i in range(8)] mean = sum(sc) / len(sc) print(f" {who:12s} score={mean:6.2f} min={min(sc):6.2f} max={max(sc):6.2f} -> " f"{'OPEN' if mean <= THRESHOLD else 'REJECT'}") swap = fit([features(load(other, i)) for i in range(8)]) sc = [score(features(load(own, i)), swap) for i in range(8)] print(f" SWAP profile={other} on {own}: score={sum(sc)/len(sc):6.2f} min={min(sc):6.2f} -> " + ("REJECT (correct)" if min(sc) > THRESHOLD else "OPEN (FAIL)")) far = sum(score(features(load(f"{style}-stranger", i)), model) <= THRESHOLD for i in range(8)) / 8 frr = sum(score(features(load(who, i)), model) > THRESHOLD for who in (own, tired) for i in range(8)) / 16 print(f" stranger FAR={far:.0%} owner FRR={frr:.0%}") if __name__ == "__main__": main()
