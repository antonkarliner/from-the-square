Copy the Python block to marrow.py and run python marrow.py. No packages or genuine recordings. The report gives match, review and abstention counts and rates by input style and scenario. The fixed threshold is evaluated on separate synthetic seeds after model development. Similar real users or recordings can overlap. This is a review signal and does not unlock a vault. ```python # SPDX-License-Identifier: MIT """Marrow offline research prototype. Python 3.10+, no dependencies. Run python marrow.py to generate and evaluate synthetic cases. Run python marrow.py recordings.json to evaluate your own offline JSON. The top-level object has enroll and test arrays of recording objects: {"style":"any-name","role":"owner","events":[ {"ms":0,"type":"pointer","x":0.2,"y":0.35}, {"ms":64,"type":"click","x":0.24,"y":0.41}, {"ms":130,"type":"text","source":"dictation","added":5,"removed":0} ]} Coordinates are fractions of the viewport. Types are pointer, click, text. Possible text sources are typed, onscreen, and dictation. Store counts only. Test roles: owner, tired, stranger, mimic, swap. Omit role to score unlabeled. Enroll separately per style. Sessions with insufficient evidence abstain. The output is experimental evidence and should not unlock a vault by itself. """ import collections import json import math import random import statistics as st import sys THRESHOLD = 3.0 FLOORS = (.05, .03, .055, .065, .08, .05) SOURCES = ("typed", "onscreen", "dictation") def features(events): pointer, clicks, edits = [], [], [] last_ms = -1 for event in events: ms = event["ms"] if type(ms) is not int or ms < 0 or ms < last_ms: raise ValueError("timestamps must be sorted nonnegative integer milliseconds") last_ms = ms kind = event["type"] if kind in ("pointer", "click"): x, y = event["x"], event["y"] if not (0 <= x <= 1 and 0 <= y <= 1): raise ValueError("coordinates must be fractions between 0 and 1") (pointer if kind == "pointer" else clicks).append((ms, x, y)) elif kind == "text": added, removed = event["added"], event["removed"] if event["source"] not in SOURCES or any(type(n) is not int or n < 0 for n in (added, removed)): raise ValueError("invalid text source or edit counts") edits.append((ms, added, removed)) else: raise ValueError(f"unknown event type: {kind}") if len(pointer) < 20 or len(clicks) < 5 or len(edits) < 6: return None spans = [math.dist(a[1:], b[1:]) for a, b in zip(pointer, pointer[1:])] intervals = [b[0] - a[0] for a, b in zip(pointer, pointer[1:])] click_gaps = [b[0] - a[0] for a, b in zip(clicks, clicks[1:])] edit_gaps = [b[0] - a[0] for a, b in zip(edits, edits[1:])] edit_volume = sum(a + r for _, a, r in edits) return (math.log1p(100 * st.median(spans)), math.log1p(st.median(intervals) / 100), math.log1p(st.median(click_gaps) / 500), math.log1p(st.median(edit_gaps) / 500), math.log1p(st.median(a for _, a, _ in edits)), sum(r for _, _, r in edits) / max(1, edit_volume)) def fit(samples): def center_and_scale(column, floor): center = st.median(column) deviation = st.median(abs(value - center) for value in column) return center, max(floor, 1.4826 * deviation) if len(samples) < 5: raise ValueError("at least five usable enrollment sessions per style") return [center_and_scale(column, floor) for column, floor in zip(zip(*samples), FLOORS)] def distance(values, model): standardized = [abs(value - center) / scale for value, (center, scale) in zip(values, model)] # A fatigued user can slow down consistently without changing correction habits. return max(math.sqrt(st.mean(z * z for z in standardized)), standardized[-1]) def evaluate(enroll, test): by_style = collections.defaultdict(list) for row in enroll: values = features(row["events"]) if values is not None: by_style[row["style"]].append(values) models = {style: fit(samples) for style, samples in by_style.items()} if not models: raise ValueError("no usable enrollment") results = [] rates = collections.defaultdict(collections.Counter) for row in test: style = row["style"] role = row.get("role", "unlabeled") values = features(row["events"]) model = models.get(style) score = None if not values or not model else distance(values, model) decision = "abstain" if score is None else "match" if score <= THRESHOLD else "review" rates[f"{style}/{role}"][decision] += 1 results.append({"style":style, "role":role, "decision":decision, "score":None if score is None else round(score, 3)}) groups = {} for key, counts in rates.items(): total = sum(counts.values()) groups[key] = {decision:counts[decision] for decision in ("match", "review", "abstain")} groups[key]["match_rate"] = round(counts["match"] / total, 4) groups[key]["review_or_abstain_rate"] = round(1 - counts["match"] / total, 4) return {"threshold":THRESHOLD, "groups":groups, "sessions":results} def fixture(style, role, seed): rng = random.Random(seed) slow = style == "slow-dictation" factor = {"owner":1, "tired":1.19, "stranger":1.9, "mimic":1.12, "swap":.55}[role] noise = lambda: rng.uniform(.9, 1.1) events, ms, x, y = [], 0, .15, .35 for i in range(60): ms += int((150 if slow else 53) * factor * noise()) distance = (.009 if slow else .017) * noise() * (1.8 if role in ("stranger", "swap") else 1) x = (x + distance * math.cos(i / 11)) % 1 y = (y + distance * math.sin(i / 11)) % 1 events.append({"ms":ms,"type":"pointer","x":x,"y":y}) if i % 8 == 7: events.append({"ms":ms,"type":"click","x":x,"y":y}) ms = 0 for i in range(20): ms += int((910 if slow else 210) * factor * noise()) count = (7 if slow else 2) + rng.randrange(3) removed = count if i % (2 if role == "mimic" else 10) == 0 else 0 events.append({"ms":ms,"type":"text","source":"dictation" if slow else "typed", "added":count,"removed":removed}) return {"style":style,"role":role,"events":sorted(events, key=lambda e:e["ms"])} def synthetic(): styles = ("mouse-typing", "slow-dictation") roles = ("owner", "tired", "stranger", "mimic", "swap") enroll = [fixture(style, "owner", n + 1000 * g) for g, style in enumerate(styles) for n in range(16)] test = [fixture(style, role, 79000 + 1000 * g + 100 * r + n) for g, style in enumerate(styles) for r, role in enumerate(roles) for n in range(16)] return enroll, test if __name__ == "__main__": if len(sys.argv) > 1: with open(sys.argv[1], encoding="utf-8") as source: recordings = json.load(source) report = evaluate(recordings["enroll"], recordings["test"]) else: enroll, test = synthetic() report = evaluate(enroll, test) swapped = evaluate([fixture(style, "stranger", 20000 + n) for style in ("mouse-typing", "slow-dictation") for n in range(16)], [row for row in test if row["role"] == "owner"]) report["reverse_profile_swap"] = swapped["groups"] assert all(row["match"] == 0 for row in swapped["groups"].values()) print(json.dumps(report, indent=2)) ```
