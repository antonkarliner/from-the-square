Arbitrary input validation is outside this sample's scope. --- fizzbuzz.js --- /** * Return labels for integers 1 through n using ordered [divisor, word] rules. * Inputs are a non-negative integer n and integer-divisor/string-word pairs. * Rules and their entries are never mutated. */ export default function fizzBuzz(n, rules) { const result = []; for (let value = 1; value <= n; value += 1) { let label = ''; let matched = false; for (const [divisor, word] of rules) { if (value % divisor === 0) { label += word; matched = true; } } result.push(matched ? label : String(value)); } return result; } --- fizzbuzz.test.js --- import assert from 'node:assert/strict'; import test from 'node:test'; import fizzBuzz from './fizzbuzz.js'; test('classic rules replace multiples and combine matching words', () => { assert.deepEqual(fizzBuzz(15, [[3, 'Fizz'], [5, 'Buzz']]), [ '1', '2', 'Fizz', '4', 'Buzz', 'Fizz', '7', '8', 'Fizz', 'Buzz', '11', 'Fizz', '13', '14', 'FizzBuzz' ]); }); test('one rule and unmatched values work without classic defaults', () => { assert.deepEqual(fizzBuzz(6, [[2, 'Even']]), ['1', 'Even', '3', 'Even', '5', 'Even']); assert.deepEqual(fizzBuzz(3, [[7, 'Seven']]), ['1', '2', '3']); assert.deepEqual(fizzBuzz(1, [[3, 'Fizz'], [5, 'Buzz']]), ['1']); }); test('concatenation follows supplied order, including repeated divisors', () => { const values = fizzBuzz(30, [[3, 'Fizz'], [5, 'Buzz'], [2, 'Bang']]); assert.equal(values[5], 'FizzBang'); assert.equal(values[29], 'FizzBuzzBang'); assert.deepEqual(fizzBuzz(2, [[2, 'B'], [2, 'A']]), ['1', 'BA']); }); test('zero range and empty rule set return predictable arrays', () => { assert.deepEqual(fizzBuzz(0, [[3, 'Fizz']]), []); assert.deepEqual(fizzBuzz(3, []), ['1', '2', '3']); }); test('matching empty word is still a match, not a numeric fallback', () => { assert.deepEqual(fizzBuzz(2, [[2, '']]), ['1', '']); }); test('input rules remain unchanged and repeated calls are independent', () => { const rules = Object.freeze([Object.freeze([2, 'Even'])]); const first = fizzBuzz(2, rules); first[1] = 'changed'; assert.deepEqual(fizzBuzz(2, rules), ['1', 'Even']); }); Observed test totals: 6 tests, 6 pass, 0 fail.
