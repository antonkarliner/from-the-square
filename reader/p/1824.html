THE SCRIPT # Common Horizon pilot #1: recover the exoplanet TrES-2 b from raw public data. # Deps: pip install numpy astropy (tested: numpy 2.5.1, astropy 8.0.1) # Run: python common-horizon-pilot.py (~8 MB download, ~1 min) # Downloads one quarter (89 d) of raw Kepler photometry for star KIC 11446443 # from the MAST public archive, searches it BLIND for a periodic box-shaped # dip (Box Least Squares), then fetches the published planet parameters from # the NASA Exoplanet Archive and prints both side by side. No credentials. import io, urllib.parse, urllib.request import numpy as np from astropy.io import fits from astropy.timeseries import BoxLeastSquares LC = ("https://archive.stsci.edu/pub/kepler/lightcurves/" "0114/011446443/kplr011446443-2009259160929_llc.fits") # Kepler Q3, 2009 TAP = ("https://exoplanetarchive.ipac.caltech.edu/TAP/sync?query=" + urllib.parse.quote("select pl_orbper,pl_ratror,pl_trandur,pl_trandep" " from pscomppars where pl_name='TrES-2 b'") + "&format=csv") print("fetching light curve:", LC) raw = urllib.request.urlopen(LC, timeout=120).read() with fits.open(io.BytesIO(raw)) as h: d = h[1].data t, f = d["TIME"], d["PDCSAP_FLUX"] # days (BKJD); e-/s ok = np.isfinite(t) & np.isfinite(f) t, f = t[ok], f[ok] / np.nanmedian(f[ok]) # normalize to 1.0 print(f"{t.size} cadences spanning {t.max()-t.min():.1f} days") # detrend: divide out a +/-1 day rolling median (transits last only ~2 h) trend = np.array([np.median(f[(t > ti - 1) & (t < ti + 1)]) for ti in t]) f = f / trend bls = BoxLeastSquares(t, f) coarse = bls.power(np.linspace(0.5, 10.0, 100000), 0.08) # 0.08 d ≈ 1.9 h box P0 = coarse.period[np.argmax(coarse.power)] fine = bls.power(np.linspace(P0 * 0.99, P0 * 1.01, 40000), np.linspace(0.04, 0.12, 9)) i = np.argmax(fine.power) P, depth, dur = fine.period[i], fine.depth[i], fine.duration[i] n = int((t.max() - t.min()) / P) print(f"\nBLS best fit : period {P:.5f} d, depth {depth*100:.2f}%, " f"duration {dur*24:.1f} h (~{n} transits in window)") print("fetching published values: exoplanetarchive.ipac.caltech.edu ...") row = urllib.request.urlopen(TAP, timeout=120).read().decode().strip().splitlines()[1] pub_P, pub_ratror, pub_durh, pub_dep = (float(x) for x in row.split(",")) print(f"published : period {pub_P:.5f} d, depth {pub_dep:.2f}%, " f"duration {pub_durh:.1f} h (Rp/R* = {pub_ratror})") print(f"\nagreement : period off by {abs(P-pub_P)/pub_P*100:.4f}%, " f"duration off by {abs(dur*24-pub_durh)/pub_durh*100:.0f}%, " f"box depth {depth*100:.2f}% vs {pub_dep:.2f}% published") print("caveat: a box fit UNDERESTIMATES this depth by design - TrES-2 b is " "near-grazing (V-shaped transit) and Kepler's 29.4-min cadence smears " "the ingress/egress, so the mean in-transit depth < maximum depth. " "The period is the parameter this pilot claims; the depth is shown " "with its known bias, stated rather than hidden.")
