Marrow is a stdlib-only, offline behavioral-authentication prototype. The enrolled profile is built from OS-neutral JSONL pointer and text-event metadata. Fixed threshold: 3.25. Reproducible synthetic evaluation: stranger-in 0%, mimic-in 13.89%, owner-doubted 0%, tired-owner-doubted 0%, owner sessions rejected against a separately trained stranger profile 100%. Synthetic-only rates, not real-world security evidence. No raw text content, audio, or camera is used. Concatenate the code lines in order. ```python #!/usr/bin/env python3 """Offline, device-agnostic behavioral-authentication research prototype. Input is newline-delimited JSON. Each record is an event object, for example: {"t_ms":12,"type":"pointer","x":0.4,"y":0.5} {"t_ms":38,"type":"click","button":"primary"} {"t_ms":90,"type":"text","source":"typed","text":"hello","corrected":false} {"t_ms":180,"type":"correction","source":"dictation","kind":"replace"} The classifier intentionally uses timing, normalized pointer trajectories, event sequencing, and text-source/correction metadata; it never reads text content. This is a prototype, not a security product. """ from __future__ import annotations import argparse import json import math import random import statistics import sys from pathlib import Path # Fixed before generation/evaluation. A session is accepted only if its robust # standardized distance from the enrolled profile is <= this constant. THRESHOLD = 3.25 FEATURES = ( "pointer_speed", "pointer_turn", "pointer_pause", "click_gap", "text_gap", "correction_rate", "event_entropy", "pointer_text_lag", ) def read_jsonl(path: str): with (sys.stdin if path == "-" else open(path, encoding="utf-8")) as f: rows = [] for n, line in enumerate(f, 1): if line.strip(): try: row = json.loads(line) except json.JSONDecodeError as e: raise ValueError(f"{path}:{n}: invalid JSON: {e}") from e if not isinstance(row, dict): raise ValueError(f"{path}:{n}: each line must be an object") rows.append(row) return rows def feature(events): """Extract a small, order-sensitive but device-independent feature vector.""" if not events: raise ValueError("empty session") ev = sorted(events, key=lambda e: float(e.get("t_ms", 0))) times, types, pts, clicks, texts, corrections = [], [], [], [], [], 0 for e in ev: t = float(e.get("t_ms", 0)) if not math.isfinite(t) or t < 0: raise ValueError("timestamps must be finite nonnegative milliseconds") typ = str(e.get("type", "other")).lower() times.append(t); types.append(typ) if typ == "pointer": x, y = float(e.get("x", 0)), float(e.get("y", 0)) if not (math.isfinite(x) and math.isfinite(y)): raise ValueError("pointer coordinates must be finite") pts.append((t, x, y)) elif typ == "click": clicks.append(t) elif typ == "text": texts.append(t) if bool(e.get("corrected", False)): corrections += 1 elif typ == "correction": corrections += 1 def med(xs, fallback=0.0): return statistics.median(xs) if xs else fallback def gaps(xs): return [b-a for a,b in zip(xs, xs[1:]) if b >= a] distances, turns, pt_gaps = [], [], [] for a,b in zip(pts, pts[1:]): dt=max(1.0,b[0]-a[0]); dx=b[1]-a[1]; dy=b[2]-a[2] distances.append(math.hypot(dx,dy)*1000.0/dt); pt_gaps.append(dt) for a,b,c in zip(pts,pts[1:],pts[2:]): u=(b[1]-a[1],b[2]-a[2]); v=(c[1]-b[1],c[2]-b[2]) nu=math.hypot(*u); nv=math.hypot(*v) if nu*nv > 1e-9: turns.append(math.acos(max(-1,min(1,(u[0]*v[0]+u[1]*v[1])/(nu*nv))))) tg=gaps(texts); cg=gaps(clicks) # Event-sequence entropy is computed on coarse event classes, not content. counts={k:types.count(k) for k in set(types)} entropy=-sum((n/len(types))*math.log2(n/len(types)) for n in counts.values()) # Signed temporal association between pointer motion and text events, in # seconds. Missing channels receive a neutral value; mode-specific models # therefore learn their own baselines. lag=0.0 if pts and texts: lag=(statistics.median(texts)-statistics.median([p[0] for p in pts]))/1000.0 return { "pointer_speed": med(distances), "pointer_turn": med(turns), "pointer_pause": med(pt_gaps, 0)/1000.0, "click_gap": med(cg, 0)/1000.0, "text_gap": med(tg, 0)/1000.0, "correction_rate": corrections/max(1,len(texts)), "event_entropy": entropy, "pointer_text_lag": lag, } def mode_of(events): sources={str(e.get("source","unknown")).lower() for e in events if e.get("type")=="text"} return "+".join(sorted(sources)) if sources else "no-text" def load_sessions(path): """Accept either one event per line or {label,session,events} rows.""" rows=read_jsonl(path) if rows and "events" in rows[0]: return [(r.get("label"), r.get("session", str(i)), r["events"]) for i,r in enumerate(rows)] # Consecutive session records can be grouped with a session field. grouped={} for i,e in enumerate(rows): sid=str(e.get("session", "default")); grouped.setdefault(sid,[]).append(e) return [(None,sid,ev) for sid,ev in grouped.items()] def fit(sessions): bymode={} for label,sid,events in sessions: bymode.setdefault(mode_of(events),[]).append(feature(events)) model={"threshold":THRESHOLD,"profiles":{}} for mode, rows in bymode.items(): center={k:statistics.median(r[k] for r in rows) for k in FEATURES} # MAD floor avoids division by zero while preserving scale robustness. scale={k:max(1.4826*statistics.median(abs(r[k]-center[k]) for r in rows), 0.02) for k in FEATURES} model["profiles"][mode]={"center":center,"scale":scale,"n":len(rows)} return model def score(model, events): mode=mode_of(events) p=model.get("profiles",{}).get(mode) if not p: return {"accepted":False,"distance":None,"mode":mode,"reason":"untrained input style"} f=feature(events) # Average winsorized robust z-distance prevents one missing/noisy sensor # channel from dominating a decision. zs=[min(8.0,abs(f[k]-p["center"][k])/p["scale"][k]) for k in FEATURES] d=math.sqrt(sum(z*z for z in zs)/len(zs)) return {"accepted":d<=float(model.get("threshold",THRESHOLD)),"distance":round(d,4),"mode":mode} def save_model(model,path): Path(path).write_text(json.dumps(model,indent=2,sort_keys=True)+"\n",encoding="utf-8") def synthetic(seed=917): """Three input styles; each has owner, tired owner, stranger, mimic.""" rng=random.Random(seed); out=[] styles={"typed":("typed",True),"dictation":("dictation",True),"switch":("on-screen",False)} personas={"owner":(1.0,1.0),"tired-owner":(1.07,1.10),"stranger":(1.75,0.62),"mimic":(1.03,0.92)} for style,(source,pointer) in styles.items(): for person,(speed,regularity) in personas.items(): for j in range(12): t=0.0; ev=[]; x=y=0.5 n=48 for i in range(n): if pointer: t+=rng.gauss(240*regularity,35*regularity) # Mimic copies aggregate speed but not the owner's fine-grained # cross-modal rhythm, direction reversals, and correction pattern. dx=rng.gauss(0,0.018*speed); dy=rng.gauss(0,0.014*speed) ```
