A Python-standard-library prototype for listing 65. It reads timestamped, normalized motion/click/edit events, does not collect input, and emits match, step_up or insufficient. It uses two invented input styles: mouse/typed and assisted pointer/onscreen or software-inserted text. Take the Python block below, followed immediately by the Python block in this author’s “Marrow demo source part 2 of 2” reply. Save them as marrow_demo.py and run python marrow_demo.py. Both blocks together are UTF-8/LF and have SHA-256 cc549741927e227abe25298a5808e76dfe8a866c832076c59f0497e4fb465b24. CLI help documents the JSON format and offline enrollment/probe command. Use --fixtures demo.json to generate the complete synthetic corpus. With fixed threshold 2.40, in each input style: 0/24 strangers and 0/24 swapped-profile probes were accepted, and 0/24 tired-owner probes required step-up. The similar synthetic group produced 6/24 matches on mouse and 13/24 matches on assisted input. These are fixture results only. Enrollment on each intended way of using a device is needed, and a match should be one signal in a step-up flow. License: MIT, full text in the source continuation. ```python # SPDX-License-Identifier: MIT """Offline behavior-verification demonstration using invented event logs. Run: python marrow_demo.py Generate inspectable fixtures: python marrow_demo.py --fixtures demo.json Score a recording: python marrow_demo.py --enroll sessions.json --probe session.json Each recording is a JSON array of events in nondecreasing millisecond order. Pointer event: {"t":20,"kind":"move","x":0.12,"y":0.5}. Click uses the same fields and "kind":"click"; record button-down only. Text event: {"t":40,"kind":"text","source":"typed","op":"insert","count":2}. Allowed sources: typed, onscreen, inserted. Allowed ops: insert, delete. Use count for number of characters changed. Do not store the characters. Enrollment is a JSON array of two or more recordings from the same person. """ import argparse from bisect import bisect_right import json import math import random import statistics from pathlib import Path # Fixed scoring protocol. No held-out sessions affect these constants. THRESHOLD = 2.40 MIN_EVENTS = 8 FEATURES = ( ("pointer_time", 0.21), ("pointer_turn", 0.19), ("click_pause", 0.28), ("text_time", 0.23), ("text_burst", 0.27), ("text_correction", 0.095), ) def features(events): if not isinstance(events, list): raise ValueError("A recording must be an array") previous = -1 moves, clicks, text = [], [], [] for event in events: t = event["t"] if type(t) is not int or t < 0 or t < previous: raise ValueError("Timestamps must be nonnegative ordered milliseconds") previous = t kind = event["kind"] if kind in ("move", "click"): x, y = event["x"], event["y"] if not (0 <= x <= 1 and 0 <= y <= 1): raise ValueError("Coordinates must be fractions of the visible screen") (moves if kind == "move" else clicks).append((t, x, y)) elif kind == "text": if event["source"] not in ("typed", "onscreen", "inserted"): raise ValueError("Unknown text arrival channel") count, op = event["count"], event["op"] if type(count) is not int or count <= 0 or op not in ("insert", "delete"): raise ValueError("Invalid edit count or operation") text.append((t, count, op)) else: raise ValueError("Unknown event kind") result = {} if len(moves) >= MIN_EVENTS: gaps = [b[0] - a[0] for a, b in zip(moves, moves[1:]) if 0 < b[0] - a[0] < 2000] angles = [math.atan2(b[2] - a[2], b[1] - a[1]) for a, b in zip(moves, moves[1:]) if a[1:] != b[1:]] turns = [abs(math.remainder(b - a, math.tau)) for a, b in zip(angles, angles[1:])] if len(gaps) >= MIN_EVENTS and turns: result["pointer_time"] = math.log1p(statistics.median(gaps)) result["pointer_turn"] = statistics.median(turns) move_times = [event[0] for event in moves] pauses = [] for t, _, _ in clicks: index = bisect_right(move_times, t) - 1 if index >= 0 and t - move_times[index] < 1500: pauses.append(t - move_times[index]) if len(pauses) >= 4: result["click_pause"] = math.log1p(statistics.median(pauses)) if len(text) >= MIN_EVENTS: gaps = [b[0] - a[0] for a, b in zip(text, text[1:]) if 0 < b[0] - a[0] < 10000] inserts = [count for _, count, op in text if op == "insert"] deleted = sum(count for _, count, op in text if op == "delete") if len(gaps) >= MIN_EVENTS and inserts: result["text_time"] = math.log1p(statistics.median(gaps)) result["text_burst"] = math.log1p(statistics.mean(inserts)) result["text_correction"] = deleted / (sum(inserts) + deleted) return result def train(recordings): if len(recordings) < 2: raise ValueError("Supply two or more distinct enrollment sessions") rows = list(map(features, recordings)) profile = {} for name, floor in FEATURES: values = [row[name] for row in rows if name in row] if len(values) >= 2: center = statistics.median(values) mad = statistics.median(abs(value - center) for value in values) profile[name] = (center, max(floor, 1.4826 * mad)) return profile def verify(profile, events): observed = features(events) differences = {k: (observed[k] - center) / scale for k, (center, scale) in profile.items() if k in observed} if len(differences) < 3: return "insufficient", None rms = math.sqrt(statistics.mean(value * value for value in differences.values())) score = (rms + max(map(abs, differences.values()))) / 2 return ("match" if score <= THRESHOLD else "step_up"), round(score, 3) ```
